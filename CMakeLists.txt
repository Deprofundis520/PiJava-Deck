cmake_minimum_required(VERSION 3.16)
project(PiJavaSteamDeck VERSION 0.1.0 LANGUAGES C CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.14)
find_package(Threads REQUIRED)
add_executable(pijava-deck src/j2mevm.cpp src/puff.c)
target_compile_features(pijava-deck PRIVATE cxx_std_17)
target_include_directories(pijava-deck PRIVATE src)
target_compile_definitions(pijava-deck PRIVATE J2ME_ENABLE_SDL J2ME_VERSION="0.2.114-deck-experimental3.14")
target_compile_options(pijava-deck PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>)
target_link_libraries(pijava-deck PRIVATE PkgConfig::SDL2 Threads::Threads ${CMAKE_DL_LIBS})
install(TARGETS pijava-deck RUNTIME DESTINATION .)
install(DIRECTORY romfs DESTINATION . PATTERN "jar" EXCLUDE)
install(PROGRAMS tools/launch-steamdeck.sh DESTINATION .)
install(FILES LICENSE NOTICE.md THIRD_PARTY.md docs/STEAMDECK.md DESTINATION .)
install(DIRECTORY licenses DESTINATION .)
install(DIRECTORY games DESTINATION .)
install(FILES assets/icon.png DESTINATION .)
install(PROGRAMS "$<TARGET_FILE:pijava-deck>" DESTINATION . RENAME "PiJava")
install(PROGRAMS "$<TARGET_FILE:pijava-deck>" DESTINATION . RENAME "安装到桌面")
install(PROGRAMS "$<TARGET_FILE:pijava-deck>" DESTINATION . RENAME "一键安装")
install(PROGRAMS "$<TARGET_FILE:pijava-deck>" DESTINATION . RENAME "添加到游戏模式")
include(CTest)
if(BUILD_TESTING)
    add_executable(deck-input-test tests/deck_input_test.cpp)
    target_compile_features(deck-input-test PRIVATE cxx_std_17)
    target_include_directories(deck-input-test PRIVATE src)
    add_test(NAME deck-input COMMAND deck-input-test)
    add_executable(deck-ui-test tests/deck_ui_test.cpp)
    target_compile_features(deck-ui-test PRIVATE cxx_std_17)
    target_include_directories(deck-ui-test PRIVATE src)
    add_test(NAME deck-ui COMMAND deck-ui-test "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(deck-ui PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(deck-launcher-test tests/deck_launcher_test.cpp)
    target_compile_features(deck-launcher-test PRIVATE cxx_std_17)
    target_include_directories(deck-launcher-test PRIVATE src)
    add_test(NAME deck-launcher COMMAND deck-launcher-test)
    foreach(check navigation keymap steam ui_input games state)
        add_executable(deck-${check}-test tests/deck_${check}_test.cpp)
        target_compile_features(deck-${check}-test PRIVATE cxx_std_17)
        target_include_directories(deck-${check}-test PRIVATE src)
        add_test(NAME deck-${check} COMMAND deck-${check}-test)
    endforeach()
endif()
